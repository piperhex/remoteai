cmake_minimum_required(VERSION 3.20)
project(csw_desktop_video LANGUAGES CXX)
if(NOT MSVC OR NOT FFMPEG_SDK OR NOT VIDEO_OUTPUT)
  message(FATAL_ERROR "Windows MSVC, FFMPEG_SDK and VIDEO_OUTPUT are required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(desktop-video-core STATIC capture.cpp duplication.cpp gdi.cpp encoder.cpp scaler.cpp
  audio.cpp control.cpp desktop_context.cpp)
target_compile_definitions(desktop-video-core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(desktop-video-core PUBLIC /W4 /WX /EHsc /utf-8)
target_include_directories(desktop-video-core SYSTEM PUBLIC "${FFMPEG_SDK}/include")
target_link_libraries(desktop-video-core PUBLIC d3d11 dxgi windowsapp gdi32 user32 ole32
  "${FFMPEG_SDK}/lib/avcodec.lib" "${FFMPEG_SDK}/lib/avutil.lib" "${FFMPEG_SDK}/lib/swscale.lib")
add_executable(desktop-video main.cpp)
target_link_libraries(desktop-video PRIVATE desktop-video-core)
set_target_properties(desktop-video PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${VIDEO_OUTPUT}")
add_executable(desktop-video-fixture fixture.cpp)
target_link_libraries(desktop-video-fixture PRIVATE desktop-video-core dwmapi)
add_executable(damage-tests damage_tests.cpp)
target_compile_options(damage-tests PRIVATE /W4 /WX /EHsc /utf-8)
enable_testing()
add_test(NAME damage-policy COMMAND damage-tests)
add_executable(audio-tests audio_tests.cpp)
target_compile_options(audio-tests PRIVATE /W4 /WX /EHsc /utf-8)
add_test(NAME audio-buffer COMMAND audio-tests)
add_executable(desktop-audio-fixture audio_fixture.cpp)
target_link_libraries(desktop-audio-fixture PRIVATE desktop-video-core)
