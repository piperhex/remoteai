cmake_minimum_required(VERSION 3.20)
project(csw_desktop_privacy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(privacy-protocol-test main.cpp test_platform.cpp)
target_compile_definitions(privacy-protocol-test PRIVATE PRIVACY_LEASE_MS=400)
if(WIN32)
  add_executable(privacy-windows-test test_windows_desktop.cpp)
  target_compile_definitions(privacy-windows-test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(privacy-windows-test PRIVATE /W4 /WX /EHsc /utf-8)
  target_link_libraries(privacy-windows-test PRIVATE user32)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_executable(desktop-privacy main.cpp windows.cpp windows_driver.cpp)
  target_compile_definitions(desktop-privacy PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(desktop-privacy PRIVATE /W4 /WX /EHsc /utf-8)
  target_link_libraries(desktop-privacy PRIVATE user32 setupapi newdev shell32 advapi32 wintrust crypt32 wtsapi32)
else()
  enable_language(OBJCXX)
  add_executable(desktop-privacy main.cpp macos.mm)
  target_compile_options(desktop-privacy PRIVATE -Wall -Wextra -Werror "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
  target_link_libraries(desktop-privacy PRIVATE "-framework Cocoa" "-framework CoreGraphics")
endif()
if(PRIVACY_OUTPUT)
  set_target_properties(desktop-privacy PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${PRIVACY_OUTPUT}")
endif()
